-- Main effect IDs (orbFx)
local orbFxIds = {231, 544, 551, 552, 553, 47093}

-- Extra effect IDs
local extraEffectIds = {
    18, 19, 20, 22, 23, 24, 25, 29, 30, 31, 89, 101, 154, 212, 214, 215, 216,
    499, 500, 501, 502, 503, 504, 505, 506,
    507, 508, 509, 510, 511, 512, 513, 514, 515,
    516, 517, 518, 519, 520, 521, 522, 523, 524,
    525, 526, 527, 528, 529, 530, 531, 532, 533, 534,
    535, 536, 537, 538, 539, 540, 541, 542, 47093
}

-- Initial orb effect IDs
local initialOrbIds = {210, 514, 515, 516, 517, 518, 47093}

-- Configuration
local maxStepDistance = 8
local flags = { ignoreNonPathable = true, precision = 1 }
local orbMessage = "An orb has spawned, step on it to capture it."
local orbPauseTime = 4 -- seconds

-- Internal state
local cavebotOffByMacro = false
local orbPauseEndTime = 0

-- Helper function: check effects on a tile
local function hasEffect(tile, effectList)
    for _, fx in ipairs(tile:getEffects()) do
        for _, id in ipairs(effectList) do
            if fx:getId() == id then
                return true
            end
        end
    end
    return false
end

-- Function to resume bots
local function resumeBots()
    if cavebotOffByMacro then
        CaveBot.setOn()
        cavebotOffByMacro = false
    end
    orbPauseEndTime = 0
end

-- Main macro
Orbes = macro(200, "Gather Orbs", function()
    local playerPos = player:getPosition()
    local now = os.clock()

    -- If pause time has ended, reactivate
    if orbPauseEndTime > 0 and now >= orbPauseEndTime then
        resumeBots()
    end

    -- Search for and step on orbs
    for _, tile in ipairs(g_map.getTiles(playerPos.z)) do
        local pos = tile:getPosition()
        if getDistanceBetween(playerPos, pos) <= maxStepDistance then
            -- Normal case: orbFx + extraEffect
            if hasEffect(tile, orbFxIds) and hasEffect(tile, extraEffectIds) then
                delay(2000)
                if findPath(playerPos, pos, 15, flags) then
                    autoWalk(pos, 15, flags)
                end
            -- Special case: initial orb
            elseif hasEffect(tile, initialOrbIds) then
                delay(2000)
                if findPath(playerPos, pos, 15, flags) then
                    autoWalk(pos, 15, flags)
                end
            end
        end
    end
end)

-- Detect orb spawn message
onTextMessage(function(mode, text)
    if text:lower():find(orbMessage:lower()) then
        if Orbes.isOff() then return end

        if CaveBot.isOn() then
            CaveBot.setOff()
            cavebotOffByMacro = true
        end

        orbPauseEndTime = os.clock() + orbPauseTime
    end
end)

-- Detect capture messages to resume everything early
onTextMessage(function(mode, text)
    local lowerText = text:lower()
    if lowerText:find("you have captured") or lowerText:find("you have absorbed an orb") then
        if Orbes.isOff() or orbPauseEndTime == 0 then return end
        schedule(1000, function()
            resumeBots()
        end)
    end
end)
