local rootWidget = g_ui.getRootWidget()

-- Utility functions to interact with the Supply Stash UI
local function findWindowByTitle(widget, title)
    if widget:getText() == title then
        return widget:getId()
    end
    for _, child in ipairs(widget:getChildren()) do
        local found = findWindowByTitle(child, title)
        if found then return found end
    end
    return nil
end

local stashWindowId = findWindowByTitle(rootWidget, "Supply Stash")

local function findStashItemById(widget, id)
    if not widget then return nil end  

    local item = widget.getItem and widget:getItem()
    if item and item:getId() == id then
        return widget
    end

    for _, child in ipairs(widget:getChildren()) do
        local found = findStashItemById(child, id)
        if found then return found end
    end
    return nil
end

local function waitForWidget(parent, id, callback)
    local widget = parent:recursiveGetChildById(id)
    if widget and widget:isVisible() then
        callback(widget)
    else
        schedule(100, function()  -- retry after 100ms
            waitForWidget(parent, id, callback)
        end)
    end
end

local function withdrawFromStash(id, count)
    waitForWidget(rootWidget, "stashButton", function(openSuppliesBtn)
        -- click to open the supply stash
        if openSuppliesBtn.onClick then
            openSuppliesBtn.onClick(openSuppliesBtn)
        end

        -- wait for the supply stash window to appear
        waitForWidget(rootWidget, stashWindowId, function(suppliesStash)
            -- find the searched item widget recursively
            local stashItem = findStashItemById(suppliesStash, id)
            if stashItem then
                -- click the item
                if stashItem.onClick then
                    stashItem.onClick(stashItem)
                end

                -- wait for the retrive items window
                waitForWidget(rootWidget, "countWindow", function(countWindow)
                    local slider = countWindow:recursiveGetChildById("countScrollBar")

                    if slider and slider.setValue then
                        slider:setValue(count)
                    end

                    local okBtn = countWindow:recursiveGetChildById("buttonOk")
                    if okBtn and okBtn.onClick then
                        okBtn.onClick(okBtn)
                    end

                    -- finally close the supply stash window
                    waitForWidget(rootWidget, stashWindowId, function(suppliesStash)
                        local closeBtn = suppliesStash:recursiveGetChildById("closeButton")
                        if closeBtn and closeBtn.onClick then
                            closeBtn.onClick(closeBtn)
                        end
                    end)
                end)
            else
                local closeBtn = suppliesStash:recursiveGetChildById("closeButton")
                if closeBtn and closeBtn.onClick then
                    closeBtn.onClick(closeBtn)
                end
            end
        end)
    end)
end

local function countPlayerItems(itemId)
    local total = 0
    local player = g_game.getLocalPlayer()
    if not player then return 0 end

    for _, container in pairs(g_game.getContainers()) do
        for _, item in ipairs(container:getItems()) do
            if item:getId() == itemId then
                total = total + (item:getCount() or 1)
            end
        end
    end

    return total
end

UI.Label("Resupply Settings")
local resupplyKey = "resupplySettings"
storage[resupplyKey] = storage[resupplyKey] or {}
storage[resupplyKey].items = storage[resupplyKey].items or {}
storage[resupplyKey].lessThan = storage[resupplyKey].lessThan or 50
storage[resupplyKey].count = storage[resupplyKey].count or 100

local resupplyItems = UI.Container(function(widget, items)
    storage[resupplyKey].items = items
end, true)
resupplyItems:setHeight(65)
resupplyItems:setItems(storage[resupplyKey].items)

local resupplyCounts = setupUI([[
Panel
  height: 50

  Label
    id: lessThan
    anchors.left: itemChooser.right
    anchors.top: parent.top
    margin-top: 8
    margin-left: 8
    text: "if less than"

  BotTextEdit
    id: lessThanEditor
    anchors.right: parent.right
    anchors.top: parent.top
    margin-top: 5
    margin-left: 5
    margin-right: 5
    width: 30  

  Label
    id: count
    anchors.left: itemChooser.right
    anchors.top: lessThanEditor.bottom
    margin-top: 8
    margin-left: 8
    text: "resupply"

  BotTextEdit
    id: countEditor
    anchors.right: parent.right
    anchors.top: lessThanEditor.bottom
    margin-top: 5
    margin-left: 5
    margin-right: 5
    width: 30  
]])

resupplyCounts.lessThanEditor:setText(storage[resupplyKey].lessThan)
resupplyCounts.lessThanEditor.onTextChange = function(widget, text)
  storage[resupplyKey].lessThan = text
end

resupplyCounts.countEditor:setText(storage[resupplyKey].count)
resupplyCounts.countEditor.onTextChange = function(widget, text)
  storage[resupplyKey].count = text
end

macro(10000, "Resupply", function()
    local lessThan = tonumber(storage[resupplyKey].lessThan) or 0
    local count = tonumber(storage[resupplyKey].count) or 0
    for _, itemData in ipairs(storage[resupplyKey].items) do
        local itemId = itemData.id
        if not itemId or lessThan <= 0 or count <= 0 then return end
        local totalCount = countPlayerItems(itemId)

        if totalCount < lessThan then
            withdrawFromStash(itemId, count)
            delay(300)
        end
    end
end)
