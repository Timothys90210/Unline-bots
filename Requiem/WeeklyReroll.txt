if WeeklyRerollLoaded then return end
WeeklyRerollLoaded = true

setDefaultTab("TASK")

local function getTaskWindow()
    local root = g_ui.getRootWidget()
    for _, child in ipairs(root:getChildren()) do
        local ok, id = pcall(function() return child:getId() end)
        if ok and id == "taskWindow" then return child end
    end
    return nil
end

local function ensureWeeklyProgressWindow(tbm, win)
    if not tbm.WeeklyProgressWindow and win then
        local progressPanel = win:recursiveGetChildById("progressPanel")
        if progressPanel then
            tbm.WeeklyProgressWindow = progressPanel
        end
    end
end

local function openBoard(callback)
    for _, fn in ipairs({"show", "open", "toggle"}) do
        if modules.game_taskboard[fn] then
            pcall(function() modules.game_taskboard[fn]() end)
            break
        end
    end
    schedule(800, function()
        local win = getTaskWindow()
        if win then callback(win)
        else modules.game_textmessage.displayGameMessage("[WeeklyReroll] taskWindow not found") end
    end)
end

local function reroll(difficulty)
    local tbm = modules.game_taskboard and modules.game_taskboard.TaskBoardModule
    if not tbm then
        modules.game_textmessage.displayGameMessage("[WeeklyReroll] module not found")
        return
    end

    local function doReroll(win)
        ensureWeeklyProgressWindow(tbm, win)
        local ok, err = pcall(function() tbm.setWeeklyDifficulty(difficulty) end)
        if ok then
            modules.game_textmessage.displayGameMessage("[WeeklyReroll] difficulty " .. difficulty .. " sent")
        else
            modules.game_textmessage.displayGameMessage("[WeeklyReroll] error: " .. tostring(err))
        end
    end

    local win = getTaskWindow()
    if win then doReroll(win)
    else openBoard(function(w) doReroll(w) end) end
end

local DELIVER_IDS = {[3034]=true, [3029]=true, [9057]=true, [3032]=true, [3030]=true, [3028]=true}

local function checkAndDeliver(win)
    local tbm = modules.game_taskboard and modules.game_taskboard.TaskBoardModule
    if not tbm or not tbm.currentWeeklyData then
        modules.game_textmessage.displayGameMessage("[AutoDeliver] no weekly data")
        return
    end
    local tasks = tbm.currentWeeklyData.deliveryTasks
    if not tasks then return end

    local itemPanel = win and win:recursiveGetChildById("itemPanel")
    if not itemPanel then
        modules.game_textmessage.displayGameMessage("[AutoDeliver] itemPanel not found")
        return
    end

    local ok, children = pcall(function() return itemPanel:getChildren() end)
    if not ok or not children then return end

    for _, task in pairs(tasks) do
        if DELIVER_IDS[task.clientId] then
            modules.game_textmessage.displayGameMessage("[AutoDeliver] Match: " .. tostring(task.name) .. " slot=" .. tostring(task.slot))
            local widget = children[task.slot]
            if widget then
                local deliverBtn = widget:recursiveGetChildById("deliverButton")
                if deliverBtn then
                    pcall(function() deliverBtn:onClick() end)
                    modules.game_textmessage.displayGameMessage("[AutoDeliver] delivered: " .. tostring(task.name))
                else
                    modules.game_textmessage.displayGameMessage("[AutoDeliver] deliverButton not found in slot " .. tostring(task.slot))
                end
            end
        end
    end
end

UI.Separator()

UI.Button("Auto Deliver", function()
    local tbm = modules.game_taskboard and modules.game_taskboard.TaskBoardModule
    if not tbm then
        modules.game_textmessage.displayGameMessage("[AutoDeliver] module not found")
        return
    end

    local function doRerollAndDeliver(win)
        ensureWeeklyProgressWindow(tbm, win)
        local ok, err = pcall(function() tbm.setWeeklyDifficulty(0) end)
        if not ok then
            modules.game_textmessage.displayGameMessage("[AutoDeliver] reroll error: " .. tostring(err))
            return
        end
        modules.game_textmessage.displayGameMessage("[AutoDeliver] rerolled, checking tasks...")
        schedule(800, function()
            pcall(function() tbm.updateWeekly() end)
            schedule(200, function() checkAndDeliver(getTaskWindow()) end)
        end)
    end

    local win = getTaskWindow()
    if win then doRerollAndDeliver(win)
    else openBoard(function(w) doRerollAndDeliver(w) end) end
end)
